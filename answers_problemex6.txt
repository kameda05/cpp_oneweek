問題 6-1　抽象クラスへのまとめ直し
------------------------------------------------------------------
【解答】
// airplane.h
#ifndef AIRPLANE_H
#define AIRPLANE_H

#include <iostream>
#include <string>

using namespace std;

//    飛行機クラス（抽象クラス）
class AirPlane{
protected:
    string m_type;
public:
    //    デストラクタ
    virtual ~AirPlane();
    //    タイプの取得
    string getType();
    //    飛行する（純粋仮想関数）
    virtual void fly() = 0;
};

#endif // AIRPLANE_H

// airplane.cpp
#include "airplane.h"

//    デストラクタ
AirPlane::~AirPlane()
{
}
//    タイプの取得
string AirPlane::getType()
{
    return m_type;
}

// fighteraircraft.h
#ifndef FIGHTERAIRCRAFT_H
#define FIGHTERAIRCRAFT_H

#include "airplane.h"

//    戦闘機クラス
class FighterAircraft : public AirPlane{
public:
    //    コンストラクタ
    FighterAircraft();
    //    飛行する
    void fly() override;
    //    戦闘する
    void fight();
};

#endif // FIGHTERAIRCRAFT_H

// fighteraircraft.cpp
#include "fighteraircraft.h"

//    コンストラクタ
FighterAircraft::FighterAircraft()
{
    m_type = "戦闘機";
}
//    飛行する
void FighterAircraft::fly()
{
    cout << "攻撃に出るために飛行" << endl;
}
//    戦闘する
void FighterAircraft::fight()
{
    cout << "戦闘します" << endl;
}

// passengerplane.h
#ifndef PASSENGERPLANE_H
#define PASSENGERPLANE_H

#include "airplane.h"

//    旅客機クラス
class PassengerPlane : public AirPlane{
public:
    //    コンストラクタ
    PassengerPlane();
    //    飛行する
    void fly() override;
    //    乗客を運ぶ
    void carryPassengers();
};

#endif // PASSENGERPLANE_H

// passengerplane.cpp
#include "passengerplane.h"

//    コンストラクタ
PassengerPlane::PassengerPlane()
{
    m_type = "旅客機";
}
//    飛行する
void PassengerPlane::fly()
{
    cout << "乗客を乗せて目的地まで飛行します。" << endl;
}
//    乗客を運ぶ
void PassengerPlane::carryPassengers()
{
    cout << "乗客を目的地まで運びます" << endl;
}

// main.cpp
#include <iostream>
#include "fighteraircraft.h"
#include "passengerplane.h"

using namespace std;

int main(){
    //    戦闘機クラスのインスタンスの生成
    FighterAircraft* fighter = new FighterAircraft();
    PassengerPlane* airliner = new PassengerPlane();
    //    戦闘機クラスの処理
    fighter->fly();                    //    飛行する
    fighter->fight();                //    戦闘する
    //    旅客機クラスの処理
    airliner->fly();                //    飛行する
    airliner->carryPassengers();    //    乗客を運ぶ
    //    delete処理
    delete fighter;
    delete airliner;
    return 0;
}

【解説】
2つのクラスに共通するメンバは、次の3つです。
  ・メンバ変数 m_type
  ・メンバ関数 getType()（処理も同じ）
  ・メンバ関数 fly()（名前は同じだが、処理はクラスごとに異なる）
処理が同じ m_type と getType() は、親クラス AirPlane にまとめて1つだけ書きます。
m_type は子クラスのコンストラクタで値を設定するので、private ではなく protected にします。
処理がクラスごとに異なる fly() は、AirPlane では「virtual void fly() = 0;」と
純粋仮想関数として宣言だけをし、処理は子クラスで定義（オーバーライド）します。
純粋仮想関数を持つクラスは抽象クラスとなり、インスタンスを生成できなくなります。
子クラスの fly() に付けた override は、親クラスの仮想関数をオーバーライドしていることを
コンパイラに確認させるキーワードです（関数名や引数を間違えるとエラーになります）。
また、親クラスのポインタで delete しても子クラスのデストラクタが正しく呼ばれるよう、
継承される前提のクラスには、仮想デストラクタ（virtual ~AirPlane()）を付けておきます。
main.cpp は、最後に return 0; を加えたほかは変更していません。

問題 6-2　派生クラスの作成
------------------------------------------------------------------
【解答】
// baseballplayer.h
#ifndef BASEBALLPLAYER_H
#define BASEBALLPLAYER_H

#include <iostream>
#include <string>

using namespace std;

class BaseballPlayer{
protected:
    //    背番号
    int m_number;
    //    プレイヤーの種類
    string m_kind;
    //    名前
    string m_name;
public:
    //    デストラクタ
    virtual ~BaseballPlayer();
    //    背番号の取得
    int getNumber();
    //    種類の取得
    string getKind();
    //    名前の取得
    string getName();
    //    プレイ（純粋仮想関数)
    virtual void play()=0;
};

#endif // BASEBALLPLAYER_H

// baseballplayer.cpp
#include "baseballplayer.h"

//    デストラクタ
BaseballPlayer::~BaseballPlayer()
{
}
//    背番号の取得
int BaseballPlayer::getNumber()
{
    return m_number;
}
//    種類の取得
string BaseballPlayer::getKind()
{
    return m_kind;
}
//    名前の取得
string BaseballPlayer::getName()
{
    return m_name;
}

// catcher.h
#ifndef CATCHER_H
#define CATCHER_H

#include "baseballplayer.h"

class Catcher : public BaseballPlayer{
public:
    //    コンストラクタ
    Catcher(string name,int number);
    //    キャッチャーがプレイする
    void play();
};

#endif // CATCHER_H

// catcher.cpp
#include "catcher.h"

Catcher::Catcher(string name,int number)
{
    m_name = name;
    m_number = number;
    m_kind = "キャッチャー";
}
//    キャッチャーがプレイする
void Catcher::play()
{
    cout << "ピッチャーの球を受ける" << endl;
}

（fielder.h、fielder.cpp、pitcher.h、pitcher.cpp、main.cpp は問題文のまま。変更なし）

【解説】
Pitcher、Fielder と同じ形で、BaseballPlayer を継承した Catcher クラスを作ります。
  ・コンストラクタで、名前・背番号と、種類「キャッチャー」を設定する
  ・純粋仮想関数 play() をオーバーライドする（処理の内容は自由。ここでは「ピッチャーの球を受ける」）
play() を定義しないと、Catcher も抽象クラスのままになり、new Catcher(…) でエラーになります。
また、main.cpp では BaseballPlayer* 型のポインタで delete しているので、
BaseballPlayer に仮想デストラクタ（virtual ~BaseballPlayer()）を追加しています。
仮想デストラクタがないと、子クラスのデストラクタが呼ばれません（問題6-3参照）。
この問題の子クラスは、デストラクタで後始末をする必要がないので、仮想デストラクタがなくても
実行結果は変わりませんが、継承して使うクラスには付けておくのが原則です。

問題 6-3　仮想デストラクタ
------------------------------------------------------------------
【解答】
// base.h
#ifndef BASE_H
#define BASE_H

//    基底クラス
class Base{
public:
    //    コンストラクタ
    Base();
    //    デストラクタ（仮想デストラクタにする）
    virtual ~Base();
};

#endif // BASE_H

（base.cpp、derived.h、derived.cpp、main.cpp は問題文のまま。変更なし）

【解説】
base.h のデストラクタの宣言に virtual を付けるだけで、期待される実行結果になります。
デストラクタが仮想関数でない場合、delete p; では、ポインタの型（Base*）だけを見て
Base のデストラクタが呼ばれます。そのため、Derived のデストラクタが実行されず、
コンストラクタで確保した m_pData のメモリが解放されません（メモリリーク）。
デストラクタを virtual にすると、ポインタが実際に指しているオブジェクトの型（Derived）の
デストラクタが呼ばれ、その後に自動的に Base のデストラクタが呼ばれます。
（生成のときとは逆に、派生クラス → 基底クラスの順に後始末されます）
virtual はヘッダファイルの宣言にだけ付け、base.cpp の定義には付けません。
基底クラスのポインタで派生クラスのインスタンスを扱う可能性があるクラス、つまり
継承されることを前提としたクラスには、仮想デストラクタを付けておくのが原則です。
